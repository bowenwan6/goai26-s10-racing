# DrDDS runtime environment example.
#
# Load this file before starting an application:
#   set -a
#   . /path/to/drdds.env
#   set +a
#
# The application ABI and source code do not need to change.

# Topic scope file.
# If this variable is unset, DrDDS reads /opt/drdds/topic_scope.conf.
# The CPack package installs the example config under /var/opt/robot/conf/drddslib.
export DRDDS_TOPIC_SCOPE_FILE=/var/opt/robot/conf/drddslib/dds_topic_scope.conf

# DDS domain id.
# If unset, the domain id passed by the application is used.
# For compatibility, existing applications can keep domain 0.
export DRDDS_DOMAIN_ID=0

# DDS built-in discovery lease settings.
# Larger lease time keeps peers alive longer after temporary network jitter.
export DRDDS_DDS_LEASE_SEC=30
export DRDDS_DDS_LEASE_ANNOUNCE_SEC=2

# Initial discovery announcement tuning.
export DRDDS_DDS_INITIAL_ANNOUNCEMENTS_COUNT=5
export DRDDS_DDS_INITIAL_ANNOUNCEMENTS_PERIOD_NS=100000000

# Fast DDS internal thread settings.
# DRDDS_DDS_THREAD_POLICY:
#   -1 disables explicit real-time scheduling.
#   1 is SCHED_FIFO on Linux.
export DRDDS_DDS_THREAD_POLICY=-1
export DRDDS_DDS_THREAD_PRIORITY=55
export DRDDS_DDS_THREAD_AFFINITY=0
export DRDDS_DDS_THREAD_STACK_SIZE=-1

# Communication trace. Disabled by default.
# Logs are written to:
#   /var/opt/robot/log/YYYY_MMDD/drdds_<topic>_<process_name>.log
export DRDDS_TRACE_ENABLED=0
export DRDDS_TRACE_TOPICS=/LIDAR/POINTS,/FOLLOW_ENABLE
export DRDDS_TRACE_LOG_DIR=/var/opt/robot/log

# High-rate topics are sampled after the first few records.
# Set to 1 for short full-detail captures.
export DRDDS_TRACE_EVERY_N=100

# Topic scope configuration for DrDDS.
#
# Format:
#   <topic_name>=local
#   <topic_name>=multi
#
# local:
#   Host-only topic. Uses SHM + loopback UDP. It is isolated from physical NICs.
#
# multi:
#   Cross-host topic. Uses SHM + loopback UDP + configured physical NICs.
#
# If a topic is not listed here, DrDDS keeps the compatible default behavior:
#   topic scope = multi
#
# Supported aliases:
#   local aliases: localhost, host, shm
#   multi aliases: multihost, remote, udp

/LOCAL_IMU=local
/LOCAL_CAMERA=local
/LOCAL_DEBUG=local

/REMOTE_GAIT=multi
/ROBOT_STATUS=multi
/FLEET_COMMAND=multi
