# /etc/systemd/system/rslidar-dual.service
[Unit]
Description=RoboSense dual RS-LiDAR-AIRY ROS 2 driver
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
User=golai
Environment=HOME=/home/golai
Environment=ROS_DOMAIN_ID=0
Environment=RMW_IMPLEMENTATION=rmw_fastrtps_cpp
WorkingDirectory=/home/golai/goai_embodied_future_material
Environment=FASTRTPS_DEFAULT_PROFILES_FILE=/home/golai/.ros/fastdds_ethernet.xml
ExecStart=/bin/bash -lc 'source /opt/ros/jazzy/setup.bash && source /home/golai/goai_embodied_future_material/install/setup.bash && exec ros2 run rslidar_sdk rslidar_sdk_node --ros-args -p config_path:=/home/golai/goai_embodied_future_material/src/dual_airy_merger/config/airy_dual.yaml'
KillSignal=SIGINT
Restart=on-failure
RestartSec=2
KillMode=control-group
TimeoutStopSec=15
SendSIGKILL=yes

[Install]
WantedBy=multi-user.target
